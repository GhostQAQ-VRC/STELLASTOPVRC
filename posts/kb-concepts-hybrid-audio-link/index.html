位域压缩模式

[UdonSynced] byte _flags = 0b10;  // bit0=playing, bit1=locked

void CopyIntoFlags()
{
    _flags = 0;
    if (_syncOwnerPlaying) _flags |= 1;
    if (_syncLocked) _flags |= 2;
}
